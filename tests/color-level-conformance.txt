# The color level resolution chain, one row per case:
# FORCE_COLOR | NO_COLOR | api override | detected → resolved
#
# `-` means unset for the environment columns, automatic for the override,
# and never consulted for the detected column
# The Rust host suite drives every row; hosts that cannot stub detection
# cover the `-` detected rows and leave the rest to Rust

# FORCE_COLOR beats everything, including NO_COLOR and the API override
0 | set | truecolor | - → none
1 |  -  |     -     | - → basic
2 |  -  |     -     | - → ansi256
3 | set |     -     | - → truecolor

# invalid FORCE_COLOR values are treated as absent
junk | - | basic    | - → basic
4    | - | disabled | - → none

# NO_COLOR presence beats the API override
- | set | truecolor | - → none
- | set |     -     | - → none

# the API override skips detection
- | - | disabled  | - → none
- | - | basic     | - → basic
- | - | ansi256   | - → ansi256
- | - | truecolor | - → truecolor

# without any signal the detected level applies
- | - | - | basic → basic
- | - | - | ansi256 → ansi256
- | - | - | truecolor → truecolor

# terminals that cannot be detected still get full color
- | - | - | none → truecolor
